          ```javascript
          import Link from 'next/link';
          
          // 1. Adicionamos o "async" antes da função
          export default async function BlogPost({ params }) {
            // 2. Aguardamos os parâmetros carregarem
            const parametros = await params; /* @@[No Next.js 15, você precisa usar o `await` porque o `params` passou a ser uma *Promise* (uma operação assíncrona) com o objetivo de melhorar a performance da aplicação. Isso permite que o servidor comece a montar e enviar outras partes da página para o usuário antes mesmo de terminar de processar a URL. Como se trata de uma Promise, o JavaScript exige obrigatoriamente o uso do `await` para avisar ao código que ele deve esperar o dado da rota (como o seu `slug`) carregar completamente antes de tentar exibi-lo na tela.] */
          
            return (
              <div style={{ textAlign: "center", padding: '50px' }}>
                <h1>Você está lendo a página dinâmica!</h1>
                
                {/* 3. Agora usamos a variável que aguardou o resultado */}
                <h2 style={{ color: 'gray' }}>O slug acessado foi: {parametros.slug}</h2>
                
                <div style={{ marginTop: '30px' }}>
                  <Link href="/" style={{ color: 'blue', textDecoration: 'underline' }}>
                    Voltar para a Home
                  </Link>
                </div>
              </div>
            );
          }
          ```

    - error.js
      ```javascript
      'use client'; // Obrigatório em arquivos de erro!
      
      export default function Error({ error, reset }) {
        return (
          <div style={{ textAlign: 'center', padding: '50px', backgroundColor: '#ffe6e6' }}>
            <h2 style={{ color: 'red' }}>Opa! Algo deu errado. 💥</h2>
            <p>{error.message}</p>
            
            {/* Botão para tentar renderizar a rota novamente sem recarregar o site inteiro */}
            <button onClick={() => reset()} style={{ padding: '10px', marginTop: '10px' }}>
              Tentar novamente
            </button>
          </div>
        );
      }
      
      /* O `error.js` é um componente de cliente (`'use client'`) no Next.js que isola falhas em rotas específicas. Ele atua como um escudo: se uma página quebra, ele exibe uma interface de erro amigável com a opção de tentar recarregar o conteúdo (`reset()`), garantindo que o restante do site, como os layouts e menus, continue funcionando perfeitamente sem travar a aplicação inteira. */
      ```

      ```javascript
      /* O Next.js já cria um layout.js padrão quando você roda o create-next-app. Criado um menu de navegação aqui, assim, ao trocar de página, o menu nunca pisca ou recarrega. */
      
      export default function RootLayout({ children }) {
        return (
          <html lang="pt-BR">
            <body style={{ margin: 0, fontFamily: "sans-serif" }}>
              
              {/* Este menu vai aparecer em TODAS as páginas do site */}
              <nav style={{ padding: '20px', backgroundColor: '#f0f0f0', display: 'flex', gap: '15px', justifyContent: 'center' }}>
                <Link href="/">Home</Link>
                <Link href="/sobre">Sobre Nós</Link>
                <Link href="/blog/teste">Blog</Link>
                <Link href="/dashboard">Dashboard</Link>
              </nav>
              
              {/* O conteúdo de cada page.js será injetado aqui dentro */}
              <main style={{ padding: '20px' }}>
                {children}
              </main> 
              
            </body>
          </html>
        );
      }
      ```

      ```jsx
      import React, { useState } from "react";
      
      export default function Contador() {
        const [contador, setContador] = useState(0);
      
        return (
          <>
            <p>Contagem: {contador}</p>
            <button onClick={() => setContador(contador + 1)}>Incrementar</button>
            <button onClick={() => setContador(0)}>Resetar</button>
          </>
        );
      }
      ```
