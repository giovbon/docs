      ```python
      from fastapi import FastAPI
      from fastapi.middleware.cors import CORSMiddleware
      
      app = FastAPI()
      
      # Configuração de CORS para permitir que o frontend (porta 3000) faça requisições
      app.add_middleware(
          CORSMiddleware,
          allow_origins=["http://localhost:3000"], # Endereço do Next.js
          allow_credentials=True,
          allow_methods=["*"],
          allow_headers=["*"],
      )
      
      @app.get("/api/mensagem")
      def get_mensagem():
          return {
              "titulo": "FastAPI 🐍 🤝 Next.js ⚛️",
              "texto": "Esta mensagem foi gerada por um backend rápido em Python!"
          }
      ```

      - page.js
        ```javascript
        export default async function Home() {
          // Faz o fetch direto na porta do FastAPI (8000)
          const resposta = await fetch('http://127.0.0.1:8000/api/mensagem', { 
            cache: 'no-store' // Garante que a chamada seja feita em tempo real no dev
          });
          
          const dados = await resposta.json();
        
          return (
            <main style={{ padding: '2rem', fontFamily: 'sans-serif' }}>
              <h1>Integração Monorepo: Frontend 🤝 Backend (FastAPI)</h1>
              
              <div style={{ marginTop: '2rem', padding: '1rem', border: '1px solid #ccc', borderRadius: '8px' }}>
                <h2>{dados.titulo}</h2>
                <p>{dados.texto}</p>
              </div>
            </main>
          );
        }
        ```
