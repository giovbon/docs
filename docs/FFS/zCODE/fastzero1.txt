      from fastapi import FastAPI, HTTPException
      from fast_zero.schemas import UserSchema, UserResponse, UserDB, UserList, UserPublic, UserUpdate, Message
      
      app = FastAPI() 
      
      @app.get("/", summary="Endpoint raiz") 
      def read_root():
          return {'message': 'API FUNCIONANDO!'}
      
      database = [] 
      
      @app.post("/users/", status_code=201, response_model=UserResponse, summary="Cria um usuário")
      def create_user(user: UserSchema):
      
          dados_do_usuario = user.model_dump() 
      
          novo_id = len(database) + 1 
      
          user_with_id = UserDB( 
              id=novo_id,
              username=dados_do_usuario["username"],
              email=dados_do_usuario["email"],
              password=dados_do_usuario["password"]
          )
      
          database.append(user_with_id) 
          
      
          return {
              "message": "Usuário criado com sucesso",
              "user": user_with_id
          }
      
      @app.get('/users/', response_model=UserList, summary="Retorna todos os usuários")
      def read_users():
          return {'users': database} 
      
      @app.put('/users/{user_id}', response_model=UserPublic, summary="Atualiza totalmente um usuário")
      def update_user(user_id: int, user: UserSchema): 
          if user_id > len(database) or user_id < 1: 
              raise HTTPException( 
                  status_code=404, detail='User not found'
              ) 
      
          user_with_id = UserDB(**user.model_dump(), id=user_id) 
          database[user_id - 1] = user_with_id 
      
          return user_with_id
      
      @app.patch('/users/{user_id}', response_model=UserPublic, summary="Atualiza parcialmente um usuário")
      def patch_user(user_id: int, user: UserUpdate):
      
          if user_id > len(database) or user_id < 1:
              raise HTTPException(status_code=404, detail='User not found')
      
          user_atual = database[user_id - 1] 
          update_data = user.model_dump(exclude_unset=True) 
          user_atualizado = user_atual.model_copy(update=update_data) 
          database[user_id - 1] = user_atualizado 
      
          return user_atualizado
      
      @app.delete('/users/{user_id}', response_model=Message, summary="Deleta um usuário")
      def delete_user(user_id: int):
          if user_id > len(database) or user_id < 1:
              raise HTTPException(
                  status_code=404, detail='User not found'
              )
      
          del database[user_id - 1] 
      
          return {'message': 'User deleted'}

    - schemas.py
      ```python
      from typing import Optional
      from pydantic import BaseModel, EmailStr 
      
      class UserSchema(BaseModel):
          username: str
          email: EmailStr
          password: str
      
      class UserPublic(BaseModel):
          username: str
          email: EmailStr
      
      class UserResponse(BaseModel):
          message: str
          user: UserPublic
      
      class UserDB(UserSchema): 
          id: int
      
      class UserList(BaseModel):
          users: list[UserPublic]
      
      class UserUpdate(BaseModel):
          username: Optional[str] = None 
          email: Optional[EmailStr] = None
          password: Optional[str] = None
      
      class Message(BaseModel):
          message: str
      ```
