      ```python
      def calcular_total_com_desconto(pedido):
          itens = pedido.get("itens", [])
          total = sum(item.get("preco", 0.0) for item in itens)
          
          if total > 100.0:
              return total * 0.90  # Aplica 10% de desconto
              
          return total
      
      def dividir_conta(pedido, numero_pessoas):
          itens = pedido.get("itens", [])
          total = sum(item.get("preco", 0.0) for item in itens)
          
          # Se numero_pessoas for 0, o Python lançará ZeroDivisionError naturalmente
          return total / numero_pessoas
      ```

      ```python
      def validar_pedido(pedido):
          itens = pedido.get("itens", [])
          
          # 1. O pedido deve conter pelo menos 1 item
          if not itens:
              return False
              
          # 2. O valor total deve ser no mínimo R$ 20,00
          total = sum(item.get("preco", 0.0) for item in itens)
          if total < 20.0:
              return False
              
          # 3. Deve conter "endereco_entrega" e não pode ser vazio
          endereco = pedido.get("endereco_entrega", "")
          if not endereco:
              return False
              
          return True
      ```

      ```python
      from app.calculos import calcular_total_com_desconto, dividir_conta
      
      def test_calcular_total_sem_desconto(pedido_simples):
          assert calcular_total_com_desconto(pedido_simples) == 40.00
      
      def test_calcular_total_com_desconto_premium(pedido_premium):
          assert calcular_total_com_desconto(pedido_premium) == 117.00
      
      def test_deve_falhar_ao_dividir_por_zero(pedido_simples):
          with pytest.raises(ZeroDivisionError):
              dividir_conta(pedido_simples, 0)
      
      def test_dividir_conta_com_sucesso(pedido_simples):
          assert dividir_conta(pedido_simples, 2) == 20.00
      
      def test_arquivo_log_temporario(arquivo_log_temporario):
          arquivo_log_temporario.write("Testando log...")
          assert not arquivo_log_temporario.closed
      ```

      ```python
      from app.pagamento import finalizar_compra
      
      def test_deve_verificar_fraude_antes_de_cobrar(mocker, pedido_simples):
          mock_gateway = mocker.Mock()
          
          resultado = finalizar_compra(pedido_simples, mock_gateway)
          
          assert resultado == "Compra aprovada"
          
          roteiro_esperado = [
              call.verificar_fraude(pedido_simples),
              call.cobrar(pedido_simples)
          ]
          mock_gateway.assert_has_calls(roteiro_esperado, any_order=False)
      ```
